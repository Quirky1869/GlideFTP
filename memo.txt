========================================
  MEMO — GLIDEFTP : COMMANDES UTILES
========================================

--- Installation des outils ---

# Installer Wails v2
go install github.com/wailsapp/wails/v2/cmd/wails@latest

# Ajouter le GOPATH/bin au PATH (à mettre dans ~/.bashrc ou ~/.zshrc)
export PATH="$PATH:$(go env GOPATH)/bin"

# Vérifier la version de Wails
wails version


--- Dépendances Go ---

# Installer les libs FTP et SFTP
go get github.com/jlaffaye/ftp github.com/pkg/sftp golang.org/x/crypto/ssh

# Compiler le backend Go (vérification)
go build ./...


--- Dépendances frontend ---

cd frontend
npm install

--- Ajouter la variable GOPATH a fish (à faire une fois)---

fish_add_path (go env GOPATH)/bin

--- Build ---

# Build production Linux (nécessite webkit2gtk-4.1 sur ce système)
wails build -tags webkit2_41
# Binaire généré : build/bin/GlideFTP

# Build production Windows (nécessite mingw-w64-gcc installé)
CC=x86_64-w64-mingw32-gcc CGO_ENABLED=1 GOOS=windows \
  wails build -platform windows/amd64
# Binaire généré : build/bin/GlideFTP.exe

# Installer le compilateur croisé Windows (Arch/Manjaro)
sudo pacman -S mingw-w64-gcc

# Script tout-en-un (Linux + Windows) — voir build.sh à la racine
./build.sh            # compile les deux plateformes
./build.sh linux      # Linux seulement  → build/bin/linux/GlideFTP
./build.sh windows    # Windows seulement → build/bin/windows/GlideFTP.exe


--- Lancer l'application ---

./build/bin/linux/GlideFTP


--- Développement (hot reload) ---

wails dev -tags webkit2_41


--- Frontend seul ---

cd frontend
npm run build     # build de production
npm run dev       # serveur de dev Vite (sans le backend Go)


--- Note système ---

# Ce système a webkit2gtk-4.1 (pas 4.0), d'où le tag -tags webkit2_41
# Sans ce tag, le build échoue avec "webkit2gtk-4.0 not found"
# Vérifier les versions webkit disponibles :
pkg-config --list-all | grep -i webkit
